# Define expression bins
bin_expression <- function(x) {
  case_when(
    is.na(x)                 ~ NA_character_,
    x == 0                   ~ "0",
    x > 0 & x < 1             ~ "<1",
    x >= 1 & x < 10           ~ "1–10",
    x >= 10 & x < 30          ~ "10–30",
    x >= 30 & x <= 100        ~ "30–100",
    x > 100                   ~ ">100"
  )
}

# Set the desired order of the bins

  "0", "<1", "1–10",
  "10–30", "30–100", ">100"
)
