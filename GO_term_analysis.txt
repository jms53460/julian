GO_long <- imap_dfr(
  GO_list,
  function(df, species) {

    df %>%
      transmute(
        Gene_ID = as.character(
          .data[[grep("geneID", names(df),
                      ignore.case = TRUE,
                      value = TRUE)[1]]]
        ),
        GO_ID = as.character(GO_ID),
        GO_term = as.character(GO_term),
        GO_domain = as.character(GO_domain),
        Species = species
      )
  }
) %>%
  distinct(Species, Gene_ID, GO_ID, GO_domain, .keep_all = TRUE)

GO_BP <- GO_long %>%
  filter(GO_domain == "biological_process") %>%
  distinct(Species, Gene_ID, GO_ID)

#Check annotation coverage
GO_BP %>%
  group_by(Species) %>%
  summarise(
    Annotated_genes = n_distinct(Gene_ID),
    Unique_BP_terms = n_distinct(GO_ID),
    .groups = "drop"
  )

build_GO_profiles <- function(cutoff) {

  profiles <- list()

  for (sp in names(mean_TPM_list)) {

    mat <- mean_TPM_list[[sp]]

    for (st in selected_stages) {

      # Identify expressed genes
      expressed <- rownames(mat)[
        !is.na(mat[, st]) &
          mat[, st] >= cutoff
      ]

      # Retrieve BP annotations
      go <- GO_BP %>%
        filter(
          Species == sp,
          Gene_ID %in% expressed
        )

      # Number of expressed genes with BP annotation
      n_annotated <- n_distinct(go$Gene_ID)

      if (nrow(go) == 0 || n_annotated == 0) {

        profile <- numeric()

      } else {

        profile_table <- go %>%
          group_by(GO_ID) %>%
          summarise(
            Gene_count = n_distinct(Gene_ID),
            .groups = "drop"
          ) %>%
          mutate(
            Proportion = Gene_count / n_annotated
          )

        profile <- setNames(
          profile_table$Proportion,
          profile_table$GO_ID
        )
      }

      key <- paste(sp, st, sep = "_")

      profiles[[key]] <- list(
        Species = sp,
        Stage = st,
        Cutoff = cutoff,
        Expressed_genes = length(expressed),
        Annotated_genes = n_annotated,
        Profile = profile
      )
    }
  }

  profiles
}

GO_profiles_by_cutoff <- setNames(
  lapply(TPM_cutoffs, build_GO_profiles),
  paste0("TPM_", TPM_cutoffs)
)

jaccard_similarity <- function(a, b) {

  a <- names(a)
  b <- names(b)

  if (length(a) == 0 || length(b) == 0) {
    return(NA_real_)
  }

  length(intersect(a, b)) /
    length(union(a, b))
}


cosine_similarity <- function(a, b) {

  all_terms <- union(names(a), names(b))

  if (length(all_terms) == 0) {
    return(NA_real_)
  }

  vec_a <- setNames(
    numeric(length(all_terms)),
    all_terms
  )

  vec_b <- vec_a

  vec_a[names(a)] <- a
  vec_b[names(b)] <- b

  denominator <- sqrt(sum(vec_a^2)) *
    sqrt(sum(vec_b^2))

  if (denominator == 0) {
    return(NA_real_)
  }

  sum(vec_a * vec_b) / denominator
}

calculate_matched_similarity <- function(profiles, cutoff) {

  species_pairs <- combn(
    names(mean_TPM_list),
    2,
    simplify = FALSE
  )

  map_dfr(species_pairs, function(pair) {

    map_dfr(selected_stages, function(st) {

      profile_a <- profiles[[
        paste(pair[1], st, sep = "_")
      ]]$Profile

      profile_b <- profiles[[
        paste(pair[2], st, sep = "_")
      ]]$Profile

      tibble(
        Species1 = pair[1],
        Species2 = pair[2],
        Stage = st,
        Cutoff = cutoff,

        Jaccard = jaccard_similarity(
          profile_a,
          profile_b
        ),

        Cosine = cosine_similarity(
          profile_a,
          profile_b
        )
      )
    })
  })
}

plot_cross_stage_heatmap <- function(
    data,
    species_a,
    species_b,
    metric = "Jaccard",
    cutoff = 1,
    stage_order = c("Tetrad", "UM", "BM", "Pollen"),
    color_scale = "fixed"
) {

  if (!metric %in% c("Jaccard", "Cosine")) {
    stop("metric must be 'Jaccard' or 'Cosine'")
  }

  if (!color_scale %in% c("fixed", "minmax")) {
    stop("color_scale must be 'fixed' or 'minmax'")
  }

  plot_data <- data %>%
    filter(
      ((Species1 == species_a & Species2 == species_b) |
       (Species1 == species_b & Species2 == species_a)),
      Cutoff == cutoff
    ) %>%
    mutate(
      Stage_y = if_else(
        Species1 == species_a,
        as.character(Stage1),
        as.character(Stage2)
      ),
      Stage_x = if_else(
        Species1 == species_a,
        as.character(Stage2),
        as.character(Stage1)
      )
    )

  if (nrow(plot_data) == 0) {
    stop("No data found for this species pair and cutoff.")
  }

  # X-axis: Tetrad -> UM -> BM -> Pollen (left to right)
  plot_data$Stage_x <- factor(
    plot_data$Stage_x,
    levels = stage_order
  )

  # Y-axis: Tetrad -> UM -> BM -> Pollen (bottom to top)
  plot_data$Stage_y <- factor(
    plot_data$Stage_y,
    levels = stage_order
  )

  if (color_scale == "fixed") {
    color_limits <- c(0.2, 1)
  } else {
    color_limits <- range(plot_data[[metric]], na.rm = TRUE)

    if (diff(color_limits) == 0) {
      color_limits <- color_limits + c(-0.001, 0.001)
    }
  }

  ggplot(
    plot_data,
    aes(x = Stage_x, y = Stage_y, fill = .data[[metric]])
  ) +
    geom_tile(color = "white", linewidth = 0.8) +
    geom_text(
      aes(label = ifelse(
        is.na(.data[[metric]]),
        "",
        sprintf("%.2f", .data[[metric]])
      )),
      size = 4,
      color = "black"
    ) +
    scale_fill_gradientn(
      colors = heatmap_colors,
      limits = color_limits,
      na.value = "grey90",
      name = metric
    ) +
    scale_x_discrete(
      limits = stage_order,
      drop = FALSE
    ) +
    # Explicitly set bottom-to-top order
    scale_y_discrete(
      limits = stage_order,
      drop = FALSE
    ) +
    coord_equal() +
    labs(
      title = paste(species_a, "vs.", species_b),
      subtitle = paste(
        metric,
        "| TPM cutoff:", cutoff,
        "| Color scale:", color_scale
      ),
      x = paste(species_b, "developmental stage"),
      y = paste(species_a, "developmental stage")
    ) +
    theme_classic() +
    theme(
      axis.text = element_text(size = 11, color = "black"),
      axis.title = element_text(size = 12),
      plot.title = element_text(size = 14, face = "bold"),
      legend.title = element_text(size = 11),
      legend.text = element_text(size = 10)
    )
}

plot_all_cross_stage_heatmaps <- function(

) {

  # Get all unique species pairs
  species_pairs <- data %>%
    distinct(Species1, Species2) %>%
    arrange(Species1, Species2)

  # Generate each heatmap
  plots <- map2(
    species_pairs$Species1,
    species_pairs$Species2,
    ~ plot_cross_stage_heatmap(
      data = data,
      species_a = .x,
      species_b = .y,
      metric = metric,
      cutoff = cutoff,
      stage_order = stage_order,
      color_scale = color_scale
    ) +
      labs(title = paste(.x, "vs.", .y))
  )

  # Combine all panels
  combined_plot <- wrap_plots(plots, ncol = 3)
   combined_plot
  }

jaccard <- function(a, b) {

  a <- unique(a)
  b <- unique(b)

  denominator <- length(union(a, b))

  if (denominator == 0) {
    return(NA_real_)
  }

  length(intersect(a, b)) / denominator
}

cosine_similarity <- function(a, b) {

  denominator <- sqrt(sum(a^2)) * sqrt(sum(b^2))

  if (denominator == 0) {
    return(NA_real_)
  }

  sum(a * b) / denominator
}

# One sparse gene x GO-term matrix per species and ontology
GO_matrix <- list()

for (sp in names(TPM_list)) {

  genes <- rownames(TPM_list[[sp]])

  GO_matrix[[sp]] <- list()

  for (ont in ontologies) {

    go <- GO_long %>%
      filter(
        Species == sp,
        Ontology == ont,
        Gene_ID %in% genes
      ) %>%
      distinct(Gene_ID, GO_ID)

    terms <- sort(unique(go$GO_ID))

    if (length(terms) == 0) {
      GO_matrix[[sp]][[ont]] <- Matrix(
        0,
        nrow = length(genes),
        ncol = 0,
        sparse = TRUE,
        dimnames = list(genes, character())
      )
      next
    }

    GO_matrix[[sp]][[ont]] <- sparseMatrix(
      i = match(go$Gene_ID, genes),
      j = match(go$GO_ID, terms),
      x = 1,
      dims = c(length(genes), length(terms)),
      dimnames = list(genes, terms)
    )
  }
}

fast_bootstrap_one <- function(stage, sampled_indices) {

  # Store expressed-gene indices for each species
  expressed_idx <- list()

  for (sp in names(TPM_list)) {

    mat <- TPM_list[[sp]]

    idx <- sampled_indices[[sp]]

    sampled_mat <- mat[, idx, drop = FALSE]

    prevalence <- rowMeans(
      sampled_mat >= TPM_threshold,
      na.rm = TRUE
    )

    expressed_idx[[sp]] <- which(
      prevalence >= prevalence_threshold
    )
  }

  # Calculate GO profiles once per species and ontology
  profiles <- list()

  for (sp in names(TPM_list)) {

    profiles[[sp]] <- list()

    for (ont in ontologies) {

      A <- GO_matrix[[sp]][[ont]]

      expressed <- expressed_idx[[sp]]

      if (length(expressed) == 0 || ncol(A) == 0) {

        profiles[[sp]][[ont]] <- list(
          terms = character(),
          proportions = numeric()
        )

        next
      }

      A_expressed <- A[expressed, , drop = FALSE]

      # Number of expressed genes assigned to each GO term
      counts <- Matrix::colSums(A_expressed)

      # Number of expressed genes annotated in this ontology
      denominator <- sum(
        Matrix::rowSums(A_expressed) > 0
      )

      present <- counts > 0

      profiles[[sp]][[ont]] <- list(
        terms = names(counts)[present],
        proportions = counts[present] / denominator
      )
    }
  }

  # Compare all species pairs
  map_dfr(ontologies, function(ont) {

    map_dfr(species_pairs, function(pair) {

      a <- profiles[[pair[1]]][[ont]]
      b <- profiles[[pair[2]]][[ont]]

      shared_terms <- intersect(a$terms, b$terms)
      all_terms <- union(a$terms, b$terms)

      jac <- if (length(all_terms) == 0) {
        NA_real_
      } else {
        length(shared_terms) / length(all_terms)
      }

      # Align profile vectors by GO ID
      all_profile_terms <- union(a$terms, b$terms)

      vec_a <- setNames(
        numeric(length(all_profile_terms)),
        all_profile_terms
      )

      vec_b <- vec_a

      vec_a[a$terms] <- a$proportions
      vec_b[b$terms] <- b$proportions

      denom <- sqrt(sum(vec_a^2)) * sqrt(sum(vec_b^2))

      cos <- if (denom == 0) {
        NA_real_
      } else {
        sum(vec_a * vec_b) / denom
      }

      tibble(
        Stage = stage,
        Species1 = pair[1],
        Species2 = pair[2],
        Ontology = ont,
        Jaccard = jac,
        Cosine = cos
      )
    })
  })
}
