NCCcomp = function(Sp){
ggplot(UMI_comp[UMI_comp$Species == Sp,], aes(x = Prep_NCC, y = UMIs, color = Stage)) +
  geom_jitter(size = 0.5, width=0.35) +
  theme_minimal() + theme(axis.text.x = element_text(angle = 45, hjust = 1)) +
  labs(
    title = Sp,
    x = "",
    y = "Transcripts per pollen precursor 
    (x1000)") + theme(legend.position="none") +
    scale_color_manual(values = c("Meiosis" = "#555", "Tetrad" = "#EB1E2C", "UM" = "#F9A729", "UM Bi" = "#FD6F30", "UM Mono" = "#F9A729",
     "UM/BM" = "#F9D23C", "BM" = "#5FBB68", "BM/TP" = "#64CDCC", "BM_shed" = "#BBC9E5", "TP" = "#A4A4D5", "TP_shed" = "#BBC9E5"))
}

    labs(title = paste(Sp, "Batch", b, sep = " "), x = "log(UMIs+1,10)", y = "Count", fill = "Group") +
    theme_classic() + scale_fill_manual(values = c("Mono" = "purple3", "Bi" = "black", "NCC" = "red3", "Unclear" = "#999")) +
    theme(legend.position="none")
