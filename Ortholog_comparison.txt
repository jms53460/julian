# Calculate four-stage mean TPM for each species
mean_stage_expr <- lapply(names(stage_cols), function(species) {

  species_stages <- stage_cols[[species]]

  stage_matrix <- sapply(stage_order, function(stage) {

    cols <- species_stages[[stage]]

    rowMeans(
      ExpComp_mat[, cols, drop = FALSE],
      na.rm = TRUE
    )
  })

  colnames(stage_matrix) <- stage_order
  rownames(stage_matrix) <- rownames(ExpComp_mat)

  # rowMeans returns NaN when all contributing values are NA
  stage_matrix[!is.finite(stage_matrix)] <- NA_real_

  stage_matrix
})

names(mean_stage_expr) <- names(stage_cols)

# Transform each species' expression matrix
log_expr <- lapply(mean_stage_expr, function(x) {
  log10(x + 100)
})

# All six species pairs
species_pairs <- combn(
  names(log_expr),
  2,
  simplify = FALSE
)

# Calculate all cross-stage correlations
cross_species_cor <- map_dfr(species_pairs, function(pair) {

  species_a <- pair[1]
  species_b <- pair[2]

  # Correlation between all stages in species A and species B
  cor_matrix <- cor(
    log_expr[[species_a]],
    log_expr[[species_b]],
    use = "pairwise.complete.obs",
    method = "pearson"
  )

  # Convert to long format
  as.data.frame(as.table(cor_matrix)) %>%
    rename(
      Stage1 = Var1,
      Stage2 = Var2,
      Correlation = Freq
    ) %>%
    mutate(
      Species1 = species_a,
      Species2 = species_b
    ) %>%
    select(
      Species1,
      Species2,
      Stage1,
      Stage2,
      Correlation
    )
})

  c("#0571b0", "#92c5de", "#f7f7f7", "#f4a582", "#ca0020")

plot_expression_cor_heatmap <- function(
    data,
    species_a,
    species_b,
    color_scale = "fixed"
) {

  plot_data <- data %>%
    filter(
      Species1 == species_a,
      Species2 == species_b
    ) %>%
    mutate(
      Stage1 = factor(Stage1, levels = stage_order),
      Stage2 = factor(Stage2, levels = stage_order)
    )

  if (nrow(plot_data) == 0) {
    stop("No data found for this species pair.")
  }

  # Color scale
  if (color_scale == "fixed") {

    color_limits <- c(-1, 1)

  } else if (color_scale == "minmax") {

    color_limits <- range(
      plot_data$Correlation,
      na.rm = TRUE
    )

    if (diff(color_limits) == 0) {
      color_limits <- color_limits + c(-0.001, 0.001)
    }

  } else {
    stop("color_scale must be 'fixed' or 'minmax'")
  }

  ggplot(
    plot_data,
    aes(
      x = Stage2,
      y = Stage1,
      fill = Correlation
    )
  ) +

    geom_tile(
      color = "white",
      linewidth = 0.8
    ) +

    geom_text(
      aes(label = ifelse(
        is.na(Correlation),
        "",
        sprintf("%.2f", Correlation)
      )),
      size = 4,
      color = "black"
    ) +

    scale_fill_gradientn(
      colors = heatmap_colors,
      limits = color_limits,
      na.value = "grey90",
      name = "Pearson\ncorrelation"
    ) +

    scale_x_discrete(
      limits = stage_order,
      drop = FALSE
    ) +

    # First level is at the bottom of the y-axis
    scale_y_discrete(
      limits = stage_order,
      drop = FALSE
    ) +

    coord_equal() +

    labs(
      title = paste(species_a, "vs.", species_b),
      subtitle = paste("Color scale:", color_scale),
      x = paste(species_b, "developmental stage"),
      y = paste(species_a, "developmental stage")
    ) +

    theme_classic() +

    theme(
      axis.text = element_text(
        size = 10,
        color = "black"
      ),
      axis.text.x = element_text(
        angle = 45,
        hjust = 1
      ),
      axis.title = element_text(size = 11),
      plot.title = element_text(
        size = 13,
        face = "bold"
      ),
      plot.subtitle = element_text(size = 9),
      legend.title = element_text(size = 10),
      legend.text = element_text(size = 9)
    )
}

plot_all_expression_cor <- function(
    data,
    color_scale = "fixed",
    ncol = 3
) {

  plots <- map(species_pairs, function(pair) {

    plot_expression_cor_heatmap(
      data = data,
      species_a = pair[1],
      species_b = pair[2],
      color_scale = color_scale
    )
  })

  wrap_plots(
    plots,
    ncol = ncol
  )
}

process_orthogroups <- function(
    orthogroups,
    expr,
    min_tpm = 5,
    min_fraction = 0.05,
    min_cor = 0.7
) {

  # ------------------------------------------------------------
  # Explicit mapping between orthology columns and expression
  # matrices
  # ------------------------------------------------------------
  og_to_expr <- c(
    Zm_geneID = "maize",
    Os_geneID = "rice",
    At_geneID = "Arabidopsis",
    Sl_geneID = "tomato"
  )

  species_order <- c(
    "maize",
    "rice",
    "tomato",
    "Arabidopsis"
  )

  # Check expression matrices
  if (!all(species_order %in% names(expr))) {
    stop(
      "expr must contain: ",
      paste(species_order, collapse = ", ")
    )
  }

  # ------------------------------------------------------------
  # Expression filter for each species
  # ------------------------------------------------------------
  expressed <- list()

  for (sp in species_order) {

    x <- as.matrix(expr[[sp]])

    n_required <- ceiling(ncol(x) * min_fraction)

    keep <- rowSums(
      x >= min_tpm,
      na.rm = TRUE
    ) >= n_required

    expressed[[sp]] <- keep
    names(expressed[[sp]]) <- rownames(x)
  }

  # ------------------------------------------------------------
  # Containers
  # ------------------------------------------------------------
  result_matrices <- list()
  metadata_list <- list()
  diagnostics_list <- list()

  # ------------------------------------------------------------
  # Process each orthogroup
  # ------------------------------------------------------------
  for (i in seq_along(orthogroups)) {

    og <- orthogroups[[i]]

    # Make sure required columns exist
    required_cols <- names(og_to_expr)

    if (!all(required_cols %in% colnames(og))) {

      diagnostics_list[[i]] <- data.frame(
        orthogroup = i,
        status = "MISSING_ORTHOLOGY_COLUMNS",
        species = NA_character_,
        n_genes = NA_integer_,
        min_cor = NA_real_,
        stringsAsFactors = FALSE
      )

      next
    }

    # Storage for this orthogroup
    species_profiles <- list()
    species_genes <- list()

    group_failed <- FALSE
    failure_status <- NA_character_
    failure_species <- NA_character_

    # ----------------------------------------------------------
    # Process each species according to the explicit mapping
    # ----------------------------------------------------------
    for (og_col in names(og_to_expr)) {

      sp <- og_to_expr[[og_col]]
      x <- as.matrix(expr[[sp]])

      # Genes listed in this orthogroup
      genes <- unique(as.character(og[[og_col]]))

      genes <- genes[
        !is.na(genes) &
        genes != ""
      ]

      # Genes actually present in expression matrix
      genes <- intersect(
        genes,
        rownames(x)
      )

      if (length(genes) == 0) {

        group_failed <- TRUE
        failure_status <- paste0(
          "NO_EXPRESSION_GENE_",
          sp
        )
        failure_species <- sp

        break
      }

      # Apply expression filter
      genes_expressed <- genes[
        expressed[[sp]][genes]
      ]

      if (length(genes_expressed) == 0) {

        group_failed <- TRUE
        failure_status <- paste0(
          "NO_EXPRESSED_GENE_",
          sp
        )
        failure_species <- sp

        break
      }

      # --------------------------------------------------------
      # One expressed gene = no correlation test needed
      # --------------------------------------------------------
      if (length(genes_expressed) == 1) {

        profile <- x[
          genes_expressed,
          ,
          drop = FALSE
        ]

        species_profiles[[sp]] <- as.numeric(profile[1, ])
        species_genes[[sp]] <- genes_expressed

      } else {

        # ------------------------------------------------------
        # Multiple expressed paralogs
        # ------------------------------------------------------
        paralog_expr <- x[
          genes_expressed,
          ,
          drop = FALSE
        ]

        cor_mat <- cor(
          t(paralog_expr),
          use = "pairwise.complete.obs",
          method = "pearson"
        )

        # Extract upper-triangle correlations
        upper_idx <- upper.tri(cor_mat)

        pairwise_cor <- cor_mat[upper_idx]

        # Require all pairwise correlations to pass
        cor_pass <- (
          length(pairwise_cor) > 0 &&
          all(
            !is.na(pairwise_cor) &
            pairwise_cor >= min_cor
          )
        )

        if (!cor_pass) {

          group_failed <- TRUE
          failure_status <- paste0(
            "LOW_PARALOG_CORRELATION_",
            sp
          )
          failure_species <- sp

          break
        }

        # Sum correlated paralogs
        species_profiles[[sp]] <- colSums(
          paralog_expr,
          na.rm = TRUE
        )

        species_genes[[sp]] <- genes_expressed
      }
    }

    # ----------------------------------------------------------
    # Failed orthogroup
    # ----------------------------------------------------------
    if (group_failed) {

      diagnostics_list[[i]] <- data.frame(
        orthogroup = i,
        status = failure_status,
        species = failure_species,
        n_genes = NA_integer_,
        min_cor = NA_real_,
        stringsAsFactors = FALSE
      )

      next
    }

    # ----------------------------------------------------------
    # Make sure every species has a profile
    # ----------------------------------------------------------
    if (!all(species_order %in% names(species_profiles))) {

      diagnostics_list[[i]] <- data.frame(
        orthogroup = i,
        status = "MISSING_SPECIES_PROFILE",
        species = NA_character_,
        n_genes = NA_integer_,
        min_cor = NA_real_,
        stringsAsFactors = FALSE
      )

      next
    }

    # ----------------------------------------------------------
    # Combine species profiles
    # ----------------------------------------------------------
    combined <- c(
      species_profiles$maize,
      species_profiles$rice,
      species_profiles$tomato,
      species_profiles$Arabidopsis
    )

    combined <- matrix(
      combined,
      nrow = 1
    )
    if (anyNA(combined) || any(!is.finite(combined))) {

      diagnostics_list[[i]] <- data.frame(
        orthogroup = i,
        status = "NA_OR_NONFINITE_EXPRESSION",
        species = NA_character_,
        n_genes = NA_integer_,
        min_cor = NA_real_,
        stringsAsFactors = FALSE
      )

      next
    }
    # Give each sample a species prefix
    colnames(combined) <- c(
      paste0(
        "maize_",
        colnames(expr$maize)
      ),

      paste0(
        "rice_",
        colnames(expr$rice)
      ),

      paste0(
        "tomato_",
        colnames(expr$tomato)
      ),

      paste0(
        "Arabidopsis_",
        colnames(expr$Arabidopsis)
      )
    )
    maize_genes <- species_genes$maize

    group_id <- paste(
      maize_genes,
      collapse = ";"
    )

    rownames(combined) <- group_id

    result_matrices[[length(result_matrices) + 1]] <- combined
    # ----------------------------------------------------------
    # Create maize-based row name
    # ----------------------------------------------------------
    maize_genes <- species_genes$maize

    group_id <- paste(
      maize_genes,
      collapse = ";"
    )

    rownames(combined) <- group_id

    # Store expression matrix
    result_matrices[[length(result_matrices) + 1]] <- combined

    # ----------------------------------------------------------
    # Metadata
    # ----------------------------------------------------------
    metadata_list[[length(metadata_list) + 1]] <- data.frame(
      orthogroup = i,
      orthogroup_id = group_id,

      maize_genes =
        paste(species_genes$maize, collapse = ";"),

      rice_genes =
        paste(species_genes$rice, collapse = ";"),

      tomato_genes =
        paste(species_genes$tomato, collapse = ";"),

      Arabidopsis_genes =
        paste(species_genes$Arabidopsis, collapse = ";"),

      n_maize =
        length(species_genes$maize),

      n_rice =
        length(species_genes$rice),

      n_tomato =
        length(species_genes$tomato),

      n_Arabidopsis =
        length(species_genes$Arabidopsis),

      stringsAsFactors = FALSE
    )

    # ----------------------------------------------------------
    # Successful orthogroup diagnostic
    # ----------------------------------------------------------
    diagnostics_list[[i]] <- data.frame(
      orthogroup = i,
      status = "PASS",
      species = NA_character_,
      n_genes = sum(
        lengths(species_genes)
      ),
      min_cor = NA_real_,
      stringsAsFactors = FALSE
    )
  }

  # ------------------------------------------------------------
  # Combine diagnostics safely
  # ------------------------------------------------------------
  diagnostics <- do.call(
    rbind,
    diagnostics_list
  )

  rownames(diagnostics) <- NULL

  # ------------------------------------------------------------
  # No groups passed
  # ------------------------------------------------------------
  if (length(result_matrices) == 0) {

    return(
      list(
        expression = NULL,
        metadata = NULL,
        diagnostics = diagnostics
      )
    )
  }

  # ------------------------------------------------------------
  # Combine all orthogroups
  # ------------------------------------------------------------
  expression_matrix <- do.call(
    rbind,
    result_matrices
  )

  metadata <- do.call(
    rbind,
    metadata_list
  )

  rownames(metadata) <- metadata$orthogroup_id

  # ------------------------------------------------------------
  # Final safety check: no NA/Inf
  # ------------------------------------------------------------
  expression_matrix[
    !is.finite(expression_matrix)
  ] <- NA_real_

  # Remove groups containing NA
  keep_rows <- complete.cases(
    expression_matrix
  )

  expression_matrix <- expression_matrix[
    keep_rows,
    ,
    drop = FALSE
  ]

  metadata <- metadata[
    metadata$orthogroup_id %in%
      rownames(expression_matrix),
    ,
    drop = FALSE
  ]

  rownames(metadata) <- metadata$orthogroup_id

  # ------------------------------------------------------------
  # Return
  # ------------------------------------------------------------
  return(
    list(
      expression = expression_matrix,
      metadata = metadata,
      diagnostics = diagnostics
    )
  )
}
