run_topGO <- function(gene_sets, GO_data, universe) {

  # Keep biological process annotations only
  GO_BP <- GO_data[
    GO_data$GO_domain == "biological_process", ]

  # Create gene-to-GO mapping
  gene2GO <- by(
    GO_BP$GO_ID,
    GO_BP[[1]],
    function(x) unique(as.character(x))
  )

  gene2GO <- as.list(gene2GO)

  # Remove genes not in the universe
  gene2GO <- gene2GO[names(gene2GO) %in% universe]

  # Run enrichment for each gene category
  results <- lapply(names(gene_sets), function(category) {

    selected_genes <- gene_sets[[category]]

    # Create named vector indicating genes of interest
    geneList <- factor(
      as.integer(universe %in% selected_genes)
    )

    names(geneList) <- universe

    # Create topGO object
    GOdata <- new(
      "topGOdata",
      ontology = "BP",
      allGenes = geneList,
      annot = annFUN.gene2GO,
      gene2GO = gene2GO,
      nodeSize = 5
    )

    # Fisher's exact test
    result <- runTest(
      GOdata,
      algorithm = "weight01",
      statistic = "fisher"
    )

    # Extract results
    GO_results <- GenTable(
      GOdata,
      weight01Fisher = result,
      topNodes = length(score(result))
    )

    # Calculate adjusted p-values
    pvals <- score(result)

    GO_results$pvalue <- pvals[GO_results$GO.ID]

    GO_results$FDR <- p.adjust(
      GO_results$pvalue,
      method = "BH"
    )

    GO_results$Category <- category

    GO_results
  })

  names(results) <- names(gene_sets)

  return(results)
}
